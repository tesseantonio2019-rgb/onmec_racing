    <script>
        document.addEventListener('DOMContentLoaded', () => {
            // 1. Menu Mobile Toggle
            const mobileMenuBtn = document.getElementById('mobile-menu-btn');
            const mobileMenu = document.getElementById('mobile-menu');
            const mobileLinks = document.querySelectorAll('.mobile-link');

            mobileMenuBtn.addEventListener('click', () => {
                mobileMenu.classList.toggle('hidden');
            });

            mobileLinks.forEach(link => {
                link.addEventListener('click', () => {
                    mobileMenu.classList.add('hidden');
                });
            });

            // Selettore stagione sponsor (2026 / 2027)
            const sponsorTabs = document.querySelectorAll('.sponsor-tab');
            const activeTab = ['bg-racing-red', 'text-white', 'shadow'];
            const inactiveTab = ['text-gray-400', 'hover:text-white'];
            const selectSponsorTab = (tab) => {
                sponsorTabs.forEach(t => {
                    const on = t === tab;
                    t.setAttribute('aria-selected', on);
                    t.tabIndex = on ? 0 : -1;
                    t.classList.remove(...(on ? inactiveTab : activeTab));
                    t.classList.add(...(on ? activeTab : inactiveTab));
                    document.getElementById(t.getAttribute('aria-controls')).classList.toggle('hidden', !on);
                });
            };
            sponsorTabs.forEach((tab, i) => {
                tab.addEventListener('click', () => selectSponsorTab(tab));
                tab.addEventListener('keydown', (e) => {
                    if (e.key !== 'ArrowRight' && e.key !== 'ArrowLeft') return;
                    const next = sponsorTabs[(i + (e.key === 'ArrowRight' ? 1 : -1) + sponsorTabs.length) % sponsorTabs.length];
                    selectSponsorTab(next);
                    next.focus();
                });
            });

            // 2. Gestione Form di Contatto senza alert()
            const contactForm = document.getElementById('contact-form');
            const formFeedback = document.getElementById('form-feedback');
            const formFeedbackText = document.getElementById('form-feedback-text');

            contactForm.addEventListener('submit', (e) => {
                e.preventDefault();
                const nameInput = document.getElementById('name').value.trim();

                formFeedbackText.textContent = `Grazie ${nameInput}! Il tuo messaggio è stato ricevuto dal team Onmec Racing. Ti ricontatteremo a breve.`;
                formFeedback.classList.remove('hidden');
                contactForm.reset();

                setTimeout(() => {
                    formFeedback.classList.add('hidden');
                }, 6000);
            });
        });
    </script>
